# in CI, where it is also audited for known vulnerabilities.

#   pip install ".[postgres,dev]" psycopg
#   pip freeze --exclude-editable | grep -vi '^cxcreditguard' > constraints.txt
# keep this header, run the test suite, and commit the result.

# binary wheel; psycopg requires the extension at exactly its own version, so
